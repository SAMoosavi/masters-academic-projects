## Setup

$$
\begin{aligned}
&\mathbb{G}: \text{q-order cyclic group},\quad P: \text{generator of } \mathbb{G} \\
&s \in \mathbb{Z}_q^*:\; P_{pub} = sP \\
&b \in \mathbb{Z}_q^*:\; T_{pub} = bP \\
&H_1: \{0,1\}^* \times \mathbb{G} \times \mathbb{G} \rightarrow \mathbb{Z}_q^* \\
&H_2: \{0,1\}^* \times \{0,1\}^* \times \mathbb{G} \rightarrow \mathbb{Z}_q^*
\end{aligned}
$$

## Pseudo-identity Generation

$$
\begin{aligned}
&t_i \in \mathbb{Z}_q^* \\
&PID_1 = t_i P,\quad AID_i = RID_i \oplus t_i T_{pub} \\
&\text{TA verifies: } RID_i = AID_i \oplus b \cdot PID_1 \\
&PID_2 = RID_i \oplus H_1(t_i T_{pub},\; PID_1,\; \Delta T)
\end{aligned}
$$

## Partial Private Key Extract

$$
\begin{aligned}
&r_i \in \mathbb{Z}_q^* \\
&R_i = \bigl(r_i + s \cdot H_1(PID,\; P_{pub})\bigr) \bmod q
\end{aligned}
$$

## Set Secret Value

$$
\begin{aligned}
&x_i \in \mathbb{Z}_q^* \\
&VPK_i = \{U_i = R_i \cdot P_{pub}\}
\end{aligned}
$$

## Signature

$$
\begin{aligned}
&w_{i1} = H_1(PID,\; U_i,\; T) \\
&w_{i2} = H_2(PID,\; m_i,\; T) \\
&X_i = x_i w_{i1} P_{pub} \\
&sg = \bigl(w_{i1} x_i + w_{i2} R_i\bigr) \bmod q \\
&\sigma_i = (X_i,\; sg)
\end{aligned}
$$

## Verification

$$
\begin{aligned}
sg \cdot P_{pub} = (w_{i1} x_i + w_{i2} R_i)P_{pub} = X_i + w_{i2} U_i
\end{aligned}
$$

## Aggregate

$$
\begin{aligned}
&X = \sum_{i=1}^n X_i,\quad SG = \sum_{i=1}^n sg_i \\
&\sigma_{agg} = (w_1, w_2, \ldots, w_n,\; X,\; SG)
\end{aligned}
$$

## Aggregate Verification

$$
\begin{aligned}
SG \cdot P_{pub} = \sum_{i=1}^n sg_i \cdot P_{pub} = \sum_{i=1}^n w_{i2} U_i + X
\end{aligned}
$$

## Traceability

$$
\begin{aligned}
RID_i = PID_2 \oplus H_1(b \cdot PID_1,\; PID_1,\; \Delta T)
\end{aligned}
$$
