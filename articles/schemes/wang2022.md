## Setup

$$
\begin{aligned}
&G_1, G_2: \text{cyclic groups of prime order } q > 2^{\nu} \\
&P, Q \in G_1,\quad e: G_1 \times G_1 \to G_2 \\
&s \in \mathbb{Z}_q^*,\quad P_{pub} = sP \\
&k \in \mathbb{Z}_q^*,\quad K = kP \\
&H_1: G_1 \to \mathbb{Z}_q^*,\quad H_2: \{0,1\}^* \times G_1 \times G_1 \to \mathbb{Z}_q^* \\
&H_3: \{0,1\}^* \times \{0,1\}^* \times G_1^6 \to \mathbb{Z}_q^*
\end{aligned}
$$

## Partial Private Key Extract

$$
\begin{aligned}
&r_i \in \mathbb{Z}_q^* \\
&R_i = r_i P \\
&k_i = H_2(PID_{i,j}, R_i) \\
&d_i = r_i + k_i \cdot s \mod q
\end{aligned}
$$

## Set Secret Value

$$
\begin{aligned}
&x_i \in \mathbb{Z}_q^* \\
&X_i = x_i P
\end{aligned}
$$

## Public Key

$$
\begin{aligned}
&PK_i = (X_i, R_i),\quad SK_i = (d_i, x_i)
\end{aligned}
$$

## Signature

$$
\begin{aligned}
&u_i \in \mathbb{Z}_q^* \\
&U_i = u_i P, \quad V_i = u_i Q \\
&h_i = H_3(m_i \| TS_i, PID_{i,j}, U_i, V_i, W_i, PK_i) \\
&W_i = (d_i + h_i \cdot x_i) Q + V_i \\
&\sigma_i = (U_i, V_i, W_i)
\end{aligned}
$$

## Aggregate

$$
\begin{aligned}
&U = \sum_{i=1}^n U_i, \quad V = \sum_{i=1}^n V_i, \quad W = \sum_{i=1}^n W_i \\
&\sigma = (U, V, W)
\end{aligned}
$$

## Verification

$$
\begin{aligned}
&\text{Single:}\quad e(W_i, P) \stackrel{?}{=} e(R_i + k_i P_{pub} + h_i X_i + U_i, Q) \\
&\text{Aggregate:}\quad e(W, P) \stackrel{?}{=} e\left(\sum_{i=1}^n R_i + \sum_{i=1}^n k_i \cdot P_{pub} + \sum_{i=1}^n h_i \cdot X_i + U, Q\right)
\end{aligned}
$$
